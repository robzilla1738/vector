async function runRepl(codeParts, options) {
	if (codeParts.includes("--help") || codeParts.includes("-h")) {
		printReplHelp();
		return;
	}

	const { trpcClient } = daemon;
	let resolvedAccount;

		resolvedAccount = await resolveAccount(trpcClient, options.account);
	} catch (error) {
		if (!isDaemonUnavailableError(error)) throw error;

		if (!spec || spec === "local" || !await hasCliAuth()) throw error;

		trpcClient,
		accountId: resolvedAccount.accountId
	} : undefined);
