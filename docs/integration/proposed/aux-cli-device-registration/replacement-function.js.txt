function getCliDeviceRegistration() {
	if (cachedCliDevice !== null) return cachedCliDevice;
	cachedCliDevice = undefined;
	const platform = process.platform === "darwin" ? "macos" : process.platform === "win32" ? "windows" : undefined;
	if (!platform) return undefined;
	try {
		const nativeMac = process.platform === "darwin";
		const status = runNativeHelperCommand(nativeMac ? "install-status" : "keys-status");
		if (status.error || !(nativeMac ? status.installed === true : status.keysInstalled)) return undefined;
		const keys = runNativeHelperCommand(nativeMac ? "get-public-keys" : "public-keys");
		if (keys.error || typeof keys.sig !== "string") return undefined;
		let installationScheme = "p256_v1";
		if (nativeMac) {
			if ((status.scheme !== "p256_v1" && status.scheme !== "pq_v1") || keys.scheme !== status.scheme) return undefined;
			installationScheme = status.scheme;
		}
		const publicKey = Buffer.from(keys.sig, "base64");
		if (nativeMac && (publicKey.toString("base64") !== keys.sig ||
			(installationScheme === "p256_v1" ? publicKey.length !== 65 || publicKey[0] !== 4 : publicKey.length !== 1952))) return undefined;
		const toBase64Url = (bytes) => Buffer.from(bytes).toString("base64url");
		const jwk = nativeMac && installationScheme === "pq_v1" ? {
			kty: "RAW",
			alg: "ML-DSA-65",
			public_key_b64: keys.sig
		} : publicKey.length === 65 && publicKey[0] === 4 ? {
			kty: "EC",
			crv: "P-256",
			x: toBase64Url(publicKey.subarray(1, 33)),
			y: toBase64Url(publicKey.subarray(33, 65)),
			ext: true,
			key_ops: ["verify"]
		} : {
			kty: "RAW",
			alg: "UNKNOWN",
			public_key_b64: keys.sig
		};
		cachedCliDevice = {
			appVersion: CLI_VERSION,
			deviceName: os.hostname() || undefined,
			platform,
			installationScheme,
			ikSigPublicKeyJwk: jwk,
			ikSigKeyFingerprint: `sha256:${createHash("sha256").update(publicKey).digest("hex")}`
		};
	} catch {}
	return cachedCliDevice;
}
