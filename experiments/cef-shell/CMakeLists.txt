cmake_minimum_required(VERSION 3.21)
project(AuxHostCandidate LANGUAGES C CXX OBJCXX)
if(NOT APPLE OR NOT CEF_ROOT)
  message(FATAL_ERROR "This development candidate requires macOS and -DCEF_ROOT=<pinned SDK directory>")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD 20)
set(PROJECT_ARCH arm64)
set(USE_SANDBOX ON CACHE BOOL "Keep Chromium subprocess sandboxing enabled" FORCE)
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
add_executable(aux-host main.mm layout-probe.cc)
add_executable(aux-host-helper helper.cc)
foreach(target aux-host aux-host-helper)
  target_include_directories(${target} PRIVATE "${CEF_ROOT}")
  target_compile_definitions(${target} PRIVATE ${CEF_COMPILER_DEFINES})
  target_link_libraries(${target} PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS})
endforeach()
